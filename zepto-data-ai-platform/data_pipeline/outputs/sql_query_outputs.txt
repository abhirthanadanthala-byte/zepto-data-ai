Query 6 - JOIN
============================================================
SELECT
            b.title,
            c.category_name,
            b.rating,
            b.price_gbp
        FROM books b
        JOIN categories c
            ON b.category_id = c.category_id
        ORDER BY b.rating DESC, b.price_gbp DESC
        LIMIT 10;
